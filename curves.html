const menuButton = document.getElementById("menuButton"); if (menuButton) { menuButton.addEventListener("click", () => { document.body.classList.toggle("nav-open"); }); } document.addEventListener("click", event => { const sidebar = document.querySelector(".sidebar"); const clickedMenu = event.target.closest("#menuButton"); const clickedSidebar = event.target.closest(".sidebar"); if (!sidebar || clickedMenu || clickedSidebar) return; if (window.innerWidth <= 900) { document.body.classList.remove("nav-open"); } });
